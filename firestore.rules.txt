// Paste this into: Firebase console -> Build -> Firestore Database -> Rules -> Publish
//
// What it does: only a signed-in user can read/write their OWN document at
// users/{their-uid}/data/state. Nobody else — signed in or not — can read
// or write it. Since each person has their own separate Firebase project,
// there will only ever be one legitimate uid here, but this keeps things
// safe even if the project URL ever leaks or a second account gets added.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId}/data/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
  }
}
